cmake_minimum_required(VERSION 3.20)
project(NuclearWarWebEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(EMSCRIPTEN)
  add_compile_options(-fexceptions)
  add_link_options(-fexceptions -sSTACK_SIZE=2097152 -sALLOW_MEMORY_GROWTH=1)
endif()
file(GLOB_RECURSE ENGINE_SOURCES CONFIGURE_DEPENDS src/game/*.cpp src/ui/*.cpp src/core/*.cpp)
add_library(nuclear_game ${ENGINE_SOURCES})
target_include_directories(nuclear_game PUBLIC include src "${CMAKE_CURRENT_SOURCE_DIR}")
include(CTest)
if(EMSCRIPTEN)
  find_program(NWR_NODE node REQUIRED)
  set(CMAKE_CROSSCOMPILING_EMULATOR "${NWR_NODE}")
endif()
if(BUILD_TESTING)
  foreach(test engine_tests ending_tests resolver_marker_tests round_plan_tests)
    add_executable(${test} tests/engine/${test}.cpp)
    target_link_libraries(${test} PRIVATE nuclear_game)
    add_test(NAME ${test} COMMAND ${test})
  endforeach()
  foreach(pair "session_tests|tests/session/session_tests.cpp" "resolver_tests|tests/resolver/resolver_core_tests.cpp" "missile_tests|tests/resolver/single_missile_tests.cpp" "player_tests|tests/frontend/tests.cpp" "revenge_tests|tests/frontend/revenge_tests.cpp" "text_tests|tests/core/game_text_tests.cpp")
    string(REPLACE "|" ";" parts "${pair}")
    list(GET parts 0 name)
    list(GET parts 1 source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE nuclear_game)
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
endif()
add_executable(state_trace reference_tools/deterministic_trace.cpp)
target_link_libraries(state_trace PRIVATE nuclear_game)
if(EMSCRIPTEN)
  set_target_properties(state_trace PROPERTIES SUFFIX ".cjs")
  target_link_options(state_trace PRIVATE -sENVIRONMENT=node)
endif()

if(EMSCRIPTEN AND BUILD_TESTING)
  foreach(target engine_tests ending_tests resolver_marker_tests round_plan_tests session_tests resolver_tests missile_tests player_tests revenge_tests text_tests)
    set_target_properties(${target} PROPERTIES SUFFIX ".cjs")
    target_link_options(${target} PRIVATE -sENVIRONMENT=node)
  endforeach()
endif()
